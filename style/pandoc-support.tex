% Pandoc's current LaTeX writer wraps automatically sized images in this macro.
% adjustbox limits width and total height while preserving the aspect ratio.
\NewDocumentCommand{\pandocbounded}{m}{%
  \begin{adjustbox}{max totalsize={\linewidth}{0.9\textheight}}%
    #1%
  \end{adjustbox}%
}
\ProvideDocumentCommand{\tightlist}{}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}%
}
