% All semantic statements share one sequence, restarted by each chapter.
% The label is established before opening tcolorbox so \label works in the body.

\NewDocumentCommand{\stepboxcounter}{m}{%
  \crefalias{semanticboxctr}{#1}%
  \refstepcounter{semanticboxctr}%
}

% The old {title} syntax remains accepted after [tcolorbox options].
\NewDocumentEnvironment{propertybox}{O{} g}{%
  \stepboxcounter{propertybox}%
  \IfNoValueTF{#2}{\def\handoutsboxtitle{\propertyname}}{%
    \ifstrempty{#2}{\def\handoutsboxtitle{\propertyname}}{\def\handoutsboxtitle{#2}}%
  }%
  \begin{tcolorbox}[semanticbox,borderline north={3pt}{0pt}{propcolor},
    title={\boxtitlewithlabel{\textcolor{propcolor}{\faBalanceScale}}{\handoutsboxtitle}{\currentboxlabel}},#1]%
}{\end{tcolorbox}}

\NewDocumentEnvironment{definitionbox}{O{} g}{%
  \stepboxcounter{definitionbox}%
  \IfNoValueTF{#2}{\def\handoutsboxtitle{\definitionname}}{%
    \ifstrempty{#2}{\def\handoutsboxtitle{\definitionname}}{\def\handoutsboxtitle{\definitionname\ \boxofname\ #2}}%
  }%
  \begin{tcolorbox}[semanticbox,borderline north={3pt}{0pt}{defcolor},
    title={\boxtitlewithlabel{\textcolor{defcolor}{\faBook}}{\handoutsboxtitle}{\currentboxlabel}},#1]%
}{\end{tcolorbox}}
